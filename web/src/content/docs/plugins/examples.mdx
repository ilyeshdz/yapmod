This is a complete, local plugin. It counts live HTTP exchanges and WebSocket messages without storing their contents, shows a badge, and removes everything when disabled.

## Package

```text
activity-badge/
├── manifest.json
├── plugin.css
└── plugin.js
```

## `manifest.json`

```json
{

  "license": "MIT",
  "api": 1,
  "tags": ["network", "badge"],
  "stylesheet": "plugin.css",
  "script": "plugin.js",
  "settings": [
    {
      "id": "show-badge",
      "label": "Show activity badge",
```

## `plugin.js`

```js
const BADGE_ID = "activity-badge";

export default {
  activate(api) {
    let httpCount = 0;
    let wsCount = 0;
    let badge = null;

    const render = () => {
      if (!api.settings.get("show-badge")) {
        badge?.remove();
        badge = null;
        return;
      }

      if (!badge) {
        badge = document.createElement("div");
        badge.id = BADGE_ID;
        badge.setAttribute("role", "status");
        badge.innerHTML = `<span data-dot></span><span data-count></span>`;
        document.body.append(badge);
      }

      badge.style.color = api.settings.get("badge-color") ?? "";
      badge.querySelector("[data-count]").textContent =
        `${httpCount} HTTP · ${wsCount} WS`;
    };

    api.onCleanup(api.http.exchanges.subscribe((exchange) => {
      httpCount += 1;

      if (api.settings.get("log-http")) {
        api.log.debug("HTTP", exchange.method, exchange.url, exchange.status);
      }

      render();
    }));

    api.onCleanup(api.webSocket.incoming.subscribe((message) => {
      wsCount += 1;

      if (api.settings.get("log-websocket")) {
        api.log.debug("WS in", message.url);
      }

      render();
    }));

    api.onCleanup(api.webSocket.outgoing.subscribe((message) => {
      wsCount += 1;

      if (api.settings.get("log-websocket")) {
        api.log.debug("WS out", message.url);
      }

      render();
    }));

    api.onCleanup(api.settings.onChange("show-badge", render));
    api.onCleanup(api.settings.onChange("badge-color", render));
    api.onCleanup(() => badge?.remove());

    render();
  },
};
```

## Why it is built this way

- It counts events and does not keep message bodies, headers or URLs in storage.
- It uses `api.onCleanup()` immediately after each subscription.
- It creates the badge only when the toggle is on.
- It updates the colour from the setting rather than relying on CSS `attr()`.
- It does not intercept or modify traffic; observation is enough for the feature.

## Install and test

1. Save the three files in a folder named `activity-badge`.
2. In yapmod, open **Settings → Yapmod → Plugins**.
3. Choose **Import plugin** and select the folder.
4. Confirm the consent dialog and enable the plugin.
5. Confirm the badge appears and counts traffic.
6. Turn the badge off in the plugin settings and confirm the element is removed.
7. Disable the plugin and confirm no badge returns when new traffic arrives.

## Example 2: Settings-driven custom panel

### `plugin.js`

```js
export default {
  activate(api) {
    const pluginId = api.id;
    let panel = null;
    let button = null;
    let isOpen = false;

    // Read the saved note from storage
    const savedNote = api.storage.get("note", "");

    // Create the floating button
    button = document.createElement("button");
    button.className = "plugin-note-taker-floating-btn";
    button.title = "Open notes";
    button.innerHTML = "+";
    button.addEventListener("click", () => {
      if (!panel) {
        // Lazy-create the panel
        panel = document.createElement("div");
        panel.className = "plugin-note-taker-panel";

        const textarea = document.createElement("textarea");
        textarea.placeholder = "Write a note...";
        textarea.value = savedNote;
        textarea.addEventListener("input", () => {
          api.storage.set("note", textarea.value);
        });

        const saveBtn = api.ui.button({
          icon: "save",
          title: "Save note",
          variant: "primary",
          size: "small",
          onClick: () => {
            api.storage.set("note", textarea.value);
            api.notify.show({ title: "Note saved", body: "Your note has been saved." });
          },
        });

        panel.append(textarea, saveBtn);
        document.body.append(panel);
      }

      isOpen = !isOpen;
      panel.classList.toggle("open", isOpen);
    });

    document.body.append(button);

    // Respond to the floating button setting
    const updateButton = () => {
      if (api.settings.get("show-floating-button")) {
        if (!document.body.contains(button)) {
          document.body.append(button);
        }
      } else {
        button.remove();
        panel?.classList.remove("open");
        isOpen = false;
      }
    };

    api.onCleanup(api.settings.onChange("show-floating-button", updateButton));
    api.onCleanup(() => button.remove());
    api.onCleanup(() => panel?.remove());
  },
};
```

### Why it is built this way

- The panel is lazy-created on first button click to avoid unnecessary DOM nodes.
- Notes are stored in `api.storage`, which is namespaced by the plugin id.
- The floating button respects the `show-floating-button` setting and updates when it changes.
- CSS is scoped under `.plugin-note-taker-*` to avoid conflicts.
- Cleanup removes both the button and the panel.

## Example 3: Network response modifier

This example shows a plugin that intercepts HTTP responses, modifies specific endpoints, and logs the changes — a common pattern for debugging or development tools.

### Package

```text
response-modifier/
├── manifest.json
└── plugin.js
```

### `manifest.json`

```json
{
  "id": "response-modifier",
  "name": "Response Modifier",
  "version": "1.0.0",
  "author": "Your name",
  "description": "Intercepts API responses and adds a custom header for debugging.",
  "license": "MIT",
  "api": 1,
  "tags": ["network", "debugging"],
  "stylesheet": "",
  "script": "plugin.js",
  "settings": [
    {
      "id": "target-url",
      "label": "Target URL pattern",
      "description": "Only modify responses whose URL contains this string.",
      "type": "text",
      "default": "/api/"
    },
    {
      "id": "enabled",
      "label": "Enable interception",
      "description": "Turn the interceptor on or off.",
      "type": "toggle",
      "default": true
    }
  ]
}
```

### `plugin.js`

```js
export default {
  activate(api) {
    let interceptorCleanup = null;

    const registerInterceptor = () => {
      // Remove the previous interceptor if it exists
      if (interceptorCleanup) {
        interceptorCleanup();
        interceptorCleanup = null;
      }

      if (!api.settings.get("enabled")) {
        return;
      }

      const targetPattern = api.settings.get("target-url") || "/api/";

      interceptorCleanup = api.http.responses.use(async (exchange) => {
        // Only modify responses matching the target pattern
        if (!exchange.url.includes(targetPattern)) {
          return undefined;
        }

        api.log.info("Intercepted response", exchange.url, exchange.status);

        // Example: wrap JSON responses with metadata
        if (exchange.responseBody && typeof exchange.responseBody === "string") {
          try {
            const parsed = JSON.parse(exchange.responseBody);
            const modified = {
              ...parsed,
              _modifiedBy: api.id,
              _interceptedAt: new Date().toISOString(),
            };

            return new Response(JSON.stringify(modified), {
              status: exchange.status,
              statusText: exchange.statusText,
              headers: { "Content-Type": "application/json" },
            });
          } catch {
            // Not JSON — pass through unchanged
            return undefined;
          }
        }

        return undefined;
      });
    };

    // Register on activation
    registerInterceptor();

    // Re-register when settings change
    api.onCleanup(api.settings.onChange("enabled", registerInterceptor));
    api.onCleanup(api.settings.onChange("target-url", registerInterceptor));

    // Clean up on disable
    api.onCleanup(() => {
      if (interceptorCleanup) {
        interceptorCleanup();
        interceptorCleanup = null;
      }
    });
  },
};
```

### Why it is built this way

- The interceptor returns `undefined` for non-matching URLs, so other plugins and the app are unaffected.
- Settings changes trigger a re-registration so the pattern and toggle take effect immediately.
- The interceptor gracefully handles non-JSON responses by passing them through.
- Logging uses `api.log` so output is namespaced and filterable.
- Cleanup removes the interceptor when the plugin is disabled, preventing orphaned handlers.
